classDiagram
    class Usuario {
        +UUID id
        +String nome
        +String email
        +String senha
        +String telefone
        +String cidade
        +String estado
        +String bio
        +Float latitude
        +Float longitude
        +cadastrar()
        +login(email, senha)
        +logout()
        +editarPerfil(dados)
        +atualizarLocalizacao(lat, lon)
    }

    class Consumidor {
        +avaliarProduto(anuncio, estrelas, comentario)
        +removerAvaliacao(anuncio)
        +iniciarChat(produtor, anuncio)
        +enviarMensagem(destinatario, conteudo)
        +buscarProdutos(termo, categoria)
        +consultarBot(pergunta)
    }

    class Produtor {
        +criarAnuncio(titulo, descricao, preco, imagem, categoria)
        +editarAnuncio(id, dados)
        +excluirAnuncio(id)
        +listarMeusAnuncios()
        +responderMensagem(destinatario, conteudo)
    }

    class Anuncio {
        +Long id
        +String titulo
        +String descricao
        +Decimal preco
        +String imagem
        +String categoria
        +Boolean ehSemente
        +Integer estoque
        +DateTime criadoEm
        +calcularMediaAvaliacoes() Float
        +contarAvaliacoes() Integer
        +calcularDistancia(usuario) Float
    }

    class Avaliacao {
        +Long id
        +Integer estrelas
        +String comentario
        +DateTime criadoEm
        +validarEstrelas() Boolean
    }

    class Mensagem {
        +Long id
        +String conteudo
        +Boolean lida
        +DateTime criadoEm
        +marcarComoLida()
    }

    class Bot {
        +String[] plantas
        +responder(pergunta) String
        +listarTopicos() String[]
    }
